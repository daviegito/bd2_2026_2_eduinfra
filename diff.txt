+            end loop;
+        end
+        $$
+        """
+    )
+
+
+def _conceder_esquemas() -> None:
+    # Gold: o dbt constrói; carga executa a opção nula; painel/análise só leem.
+    op.execute("grant usage on schema gold to eduinfra_carga, eduinfra_painel, eduinfra_analise, eduinfra_orquestrador")
+    op.execute("grant usage, create on schema gold to eduinfra_dbt")
+    op.execute("grant select on all tables in schema gold to eduinfra_carga, eduinfra_painel, eduinfra_analise")
+    op.execute("grant select on all sequences in schema gold to eduinfra_carga")
+
+    # Entrada congelada: só o orquestrador escreve; o dbt lê para o build.
+    op.execute("grant usage on schema entrada_transformacao to eduinfra_dbt, eduinfra_orquestrador")
+    op.execute("grant create on schema entrada_transformacao to eduinfra_orquestrador")
+    op.execute("grant select on all tables in schema entrada_transformacao to eduinfra_dbt, eduinfra_orquestrador")
+
+    # Fonte do CDC: o orquestrador escreve; o dbt só lê (opção nula construir-gold).
+    op.execute("grant usage on schema fonte to eduinfra_dbt, eduinfra_orquestrador")
+    op.execute("grant create on schema fonte to eduinfra_orquestrador")
+    op.execute("grant select on all tables in schema fonte to eduinfra_dbt, eduinfra_orquestrador")
+
+    # Monitoramento: a carga registra medições; o orquestrador lê e registra o
+    # frescor do pipeline. Painel e análise não alcançam a camada operacional.
+    op.execute("grant usage on schema monitoramento to eduinfra_carga, eduinfra_orquestrador")
+    op.execute("grant select, insert on monitoramento.frescor_gold to eduinfra_carga, eduinfra_orquestrador")
+    op.execute("grant usage on all sequences in schema monitoramento to eduinfra_orquestrador")
+
+    # Comparação é uma ferramenta de bancada, do papel de carga.
+    op.execute("grant usage, create on schema comparacao to eduinfra_carga")
+
+    # Camada de métricas é opcional (entrega da #98). Se existir, o painel só lê.
+    op.execute(
+        """
+        do $$
+        begin
+            if exists (select 1 from pg_namespace where nspname = 'metricas') then
+                execute 'grant usage on schema metricas to eduinfra_painel, eduinfra_analise';
+                execute 'grant select on all tables in schema metricas to eduinfra_painel, eduinfra_analise';
+            end if;
+        end
+        $$
+        """
+    )
+
+    # Privilégios padrão: objetos criados depois da migração (pelo dbt) já nascem
+    # legíveis para o painel/análise, sem um GRANT manual a cada construção.
+    op.execute(
+        "alter default privileges for role eduinfra_dbt in schema gold "
+        "grant select on tables to eduinfra_carga, eduinfra_painel, eduinfra_analise"
+    )
+    op.execute(
+        "alter default privileges for role eduinfra_orquestrador in schema entrada_transformacao "
+        "grant select on tables to eduinfra_dbt"
+    )
+    op.execute(
+        "alter default privileges for role eduinfra_orquestrador in schema fonte "
+        "grant select on tables to eduinfra_dbt"
+    )
+    op.execute(
+        "alter default privileges for role eduinfra_carga in schema monitoramento "
+        "grant select, insert on tables to eduinfra_orquestrador"
+    )
+
+
+def _criar_fila_por_uf() -> None:
+    op.execute(CRIAR_MAPA_UF)
+    op.execute("revoke all on public.acesso_uf_papel from public")
+    op.execute(CRIAR_FUNCAO_UF)
+    op.execute("revoke all on function public.uf_do_papel(name) from public")

+    op.execute(

+    )

+    op.execute(

+    )

+# Controle de acesso
+
+Responde à issue #86. Até a E3, carga, dbt, CDC, orquestrador e painel usavam
+o mesmo `POSTGRES_USER` do Compose, que é superusuário. A migração
+`0010_controle_de_acesso.py` cria papéis com privilégio mínimo, aplica recorte
+de linha na fila de vistorias e recorte de coluna na tabela `tecnico`. O
+superusuário passa a ser usado **apenas** por `migrar`/`reverter`; o
+orquestrador não recebe `DATABASE_URL_ADMIN`.
+
+## Matriz papel × objeto × operação
+
+| Papel | Lê | Escreve | Replica | Não alcança |
+| --- | --- | --- | --- | --- |
+| `eduinfra_carga` | `public` (catálogo, cadastro, vistorias, manifesto), `gold`, `monitoramento` | `INSERT` em tabelas da origem e `monitoramento.frescor_gold` | não | escrita em `gold`; `REPLICATION`; `ALTER`/`DROP` da origem |
+| `eduinfra_dbt` | `entrada_transformacao`, `fonte` (opção nula), `item_infraestrutura`, `municipio`, `escola`, `gold` | `CREATE`/`DROP`/`INSERT` só em `gold` (é o dono) | não | escrita em `public`, `entrada_transformacao`, `fonte`, `monitoramento` |
+| `eduinfra_cdc` | `vistoria`, `vistoria_item`, slot lógico | avança o slot | sim (`REPLICATION`) | escrita na origem, Gold, superusuário |
+| `eduinfra_cdc_bancada` | `vistoria`, `vistoria_item`, `escola`, `municipio`, `item_infraestrutura`, slot lógico | `INSERT` em `tecnico`, `vistoria`, `vistoria_item` | sim (`REPLICATION`) | rotina de ingestão; Gold; superusuário |
+| `eduinfra_painel` | `gold`, `metricas` (se existir) | nada | não | `public`, `entrada_transformacao`, `monitoramento` |
+| `eduinfra_analise` | `gold`, `municipio`, `escola`, colunas não pessoais de `tecnico`, `metricas` (se existir) | nada | não | `tecnico.nome`, `tecnico.matricula`, escrita, `monitoramento` |
+| `eduinfra_tecnico_df` | `fila_vistoria` e `vistoria`/`vistoria_item` da UF `DF` (RLS) | nada | não | `escola`/`municipio` direto; outras UFs |
+| `eduinfra_tecnico_sp` | `fila_vistoria` e `vistoria`/`vistoria_item` da UF `SP` (RLS) | nada | não | `escola`/`municipio` direto; outras UFs |
+| `eduinfra_orquestrador` | união de `eduinfra_carga` e `eduinfra_dbt` | `fonte`, `entrada_transformacao`; entra como `dbt` na Gold | não | `DATABASE_URL_ADMIN`; superusuário |
+
+Nenhum papel de serviço é superusuário, tem `BYPASSRLS` ou `CREATEDB`. `eduinfra_cdc`
+(consumidor de rotina) e `eduinfra_cdc_bancada` (ferramentas de bancada) têm
+`REPLICATION`; `eduinfra_carga` não tem.
+
+## Papéis e conexões no Compose
+
+| Serviço | `DATABASE_URL` | Observação |
+| --- | --- | --- |
+| `carga` | `eduinfra_carga` | `DATABASE_URL_ADMIN` (superusuário) só existe para `migrar`/`reverter` |
+| `transformacao` | `eduinfra_orquestrador` | `DATABASE_URL_TRANSFORMACAO=eduinfra_dbt` para o dbt; prepara entrada/fonte como orquestrador |
+| `cdc` | `eduinfra_cdc` | `REPLICATION` + `SELECT` nas tabelas publicadas |
+| `orquestrador` | `eduinfra_orquestrador` | pertence a carga e dbt; sem `DATABASE_URL_ADMIN` |
+| `painel` | `eduinfra_painel` | só leitura da Gold/métricas (`docker compose --profile analitico run painel`) |
+
+O `banco` continua subindo com `POSTGRES_USER` (é o bootstrap do cluster). A
+partir daí, `migrar` conecta com `DATABASE_URL_ADMIN`; todo o resto usa os
+papéis acima. Para rodar fora do Docker, copie `.env.example` e ajuste
+`EDUINFRA_SENHA_PAPEIS` (senha única local dos papéis, usada pela migração, pelo
+Compose e pela demonstração).
+
+## Recorte de linha: a fila da UF
+
+A política `vistoria_fila` filtra `vistoria` pela UF da escola via
+`public.uf_da_escola`, comparada à UF do técnico (`uf_do_papel`); `vistoria_item_fila`
+segue pela vistoria. O técnico consulta a fila por `public.fila_vistoria`, sem
+acesso direto a `escola`/`municipio`.
+
+```sql
+using (public.uf_da_escola(vistoria.co_entidade) = public.uf_do_papel(current_user))
+```
+
+A UF vem de `public.acesso_uf_papel`, um mapa papel → UF alimentado a partir de
+`tecnico.lotacao_uf`. A função `uf_do_papel` é `SECURITY DEFINER`: o técnico a
+executa, mas não lê nem altera o mapa, então não consegue trocar a própria UF.
+O mapa já nasce com `eduinfra_tecnico_df → DF` e `eduinfra_tecnico_sp → SP`;
+para um operador real, provisione o papel com a `lotacao_uf` dele.
+
+As políticas permissivas de `eduinfra_carga`/`eduinfra_orquestrador`,
+`eduinfra_dbt`/`eduinfra_cdc` (leitura) e `eduinfra_cdc_bancada` mantêm o
+pipeline e as ferramentas de bancada enxergando tudo.
+
+## Recorte de coluna: dados pessoais do técnico
+
+`eduinfra_analise` recebe `SELECT (id, lotacao_uf, ativo, registrado_em)` em
+`tecnico`. Não há `GRANT SELECT` de tabela inteira, então `nome` e `matricula`
+ficam fora do alcance do papel de análise:
+
+```sql
+grant select (id, lotacao_uf, ativo, registrado_em) on public.tecnico to eduinfra_analise;
+```

+## Testes
+
+```bash
+# Declaração: falha se um privilégio for ampliado (não precisa de banco)
+uv run pytest tests/test_controle_de_acesso.py -q -m "not integracao"
+
+# Integração: os quatro casos e os atributos dos papéis (exige banco migrado)
+uv run pytest tests/test_controle_de_acesso.py -q -m integracao
+```
+
+Os testes de declaração leem a própria migração 0010 e garantem, por exemplo,
+que `eduinfra_painel` não recebe `INSERT/UPDATE/DELETE` e que `nome`/`matricula`
+não entram no recorte de coluna. Os de integração consultam `pg_roles`, tentam
+as escritas proibidas e comparam as contagens por UF para provar a invisibilidade
+da linha. Se alguém ampliar uma permissão, os negados deixam de falhar e o teste
+quebra.

+## Limites
+
+- A camada `metricas` pertence à entrega da issue #98. Se o esquema existir, a

+"""Controle de acesso da issue #86: papéis, recorte de linha e de coluna.
+
+O comando ``demonstrar-acesso`` conecta com cada papel de privilégio mínimo e
+registra os quatro casos exigidos:
+
+1. leitura permitida  — o painel lê a Gold;
+2. escrita negada     — o painel não insere na origem;
+3. linha invisível    — o técnico de DF não vê a vistoria de SP;
+4. coluna negada      — a análise não lê ``nome``/``matricula`` de ``tecnico``.
+
+A matriz completa e as decisões estão em ``docs/controle-de-acesso.md``.
+"""
+
+import os
+from dataclasses import dataclass
+
+import psycopg
+from psycopg import errors
+from psycopg.conninfo import conninfo_to_dict, make_conninfo
+
+from eduinfra.configuracao import carregar_configuracao
+
+SENHA_PADRAO = "senha_local_dos_papeis"
+
+# Mesmos nomes da migração 0010. Mantidos aqui para o comando e os testes.
+PAPEIS = (
+    "eduinfra_carga",
+    "eduinfra_dbt",
+    "eduinfra_cdc",
+    "eduinfra_cdc_bancada",
+    "eduinfra_painel",
+    "eduinfra_analise",
+    "eduinfra_tecnico_df",
+    "eduinfra_tecnico_sp",
+    "eduinfra_orquestrador",
+)
+
+# Papéis que o Compose usa para os serviços de rotina. Nenhum é superusuário.
+PAPEIS_DE_SERVICO = (
+    "eduinfra_carga",
+    "eduinfra_dbt",
+    "eduinfra_cdc",
+    "eduinfra_painel",
+    "eduinfra_orquestrador",
+)
+
+# Colunas de tecnico que o papel de análise pode ler. nome e matricula ficam de fora.
+COLUNAS_PERMITIDAS_TECNICO = ("id", "lotacao_uf", "ativo", "registrado_em")
+
+
+@dataclass(frozen=True)
+class Caso:
+    nome: str
+    esperado: str
+    obtido: str
+    ok: bool
+
+    def linha(self) -> str:
+        situacao = "OK" if self.ok else "FALHOU"
+        return f"[{situacao}] {self.nome}: esperado {self.esperado}; obtido {self.obtido}"
+
+
+def senha_dos_papeis() -> str:
+    return os.environ.get("EDUINFRA_SENHA_PAPEIS", SENHA_PADRAO)
+
+
+def dsn_do_papel(papel: str, base: str | None = None) -> str:
+    """Troca usuário e senha da DSN base, mantendo host, porta e banco."""
+    base = base or carregar_configuracao().dsn
+    partes = conninfo_to_dict(base)
+    partes["user"] = papel
+    partes["password"] = senha_dos_papeis()
+    return make_conninfo(**partes)
+
+
+def _contar(dsn: str, consulta: str, parametros: tuple = ()) -> int:
+    with psycopg.connect(dsn, connect_timeout=5) as conexao:
+        return conexao.execute(consulta, parametros).fetchone()[0]
+
+
+def _uf_da_vistoria() -> str:
+    return """
+        select count(*)
+        from public.vistoria v
+        join public.escola e on e.co_entidade = v.co_entidade
+        join public.municipio m on m.co_municipio = e.co_municipio
+        where m.sg_uf = %s
+    """
+
+
+def _caso_leitura_permitida(base: str) -> Caso:
+    dsn = dsn_do_papel("eduinfra_painel", base)
+    try:
+        total = _contar(dsn, "select count(*) from gold.dim_escola")
+        return Caso("leitura permitida (painel -> gold)", "SELECT aceito", f"{total} linhas", True)
+    except errors.UndefinedTable:
+        # Base sem Gold construída: mostra a mesma permissão no catálogo da origem.
+        total = _contar(dsn_do_papel("eduinfra_analise", base), "select count(*) from public.municipio")
+        return Caso(
+            "leitura permitida (análise -> município)",
+            "SELECT aceito",
+            f"{total} linhas (Gold ausente)",
+            True,
+        )
+    except errors.InsufficientPrivilege:
+        return Caso("leitura permitida (painel -> gold)", "SELECT aceito", "permission denied", False)
+
+
+def _caso_escrita_negada(base: str) -> Caso:
+    dsn = dsn_do_papel("eduinfra_painel", base)
+    with psycopg.connect(dsn, connect_timeout=5) as conexao:
+        try:
+            conexao.execute(
+                "insert into public.municipio (co_municipio, no_municipio, sg_uf, co_uf) "
+                "values (-1, 'tentativa do painel', 'XX', 11)"
+            )
+        except errors.InsufficientPrivilege:
+            return Caso("escrita negada (painel -> municipio)", "permission denied", "permission denied", True)
+        conexao.rollback()
+        return Caso("escrita negada (painel -> municipio)", "permission denied", "INSERT aceito", False)
+
+
+def _caso_linha_invisivel(base: str) -> Caso:
+    total = _contar(base, "select count(*) from public.vistoria")
+    esperado_df = _contar(base, _uf_da_vistoria(), ("DF",))
+    esperado_sp = _contar(base, _uf_da_vistoria(), ("SP",))
+    sp_id = _contar(base, _uf_da_vistoria().replace("count(*)", "coalesce(max(v.id), 0)"), ("SP",))
+
+    df = dsn_do_papel("eduinfra_tecnico_df", base)
+    sp = dsn_do_papel("eduinfra_tecnico_sp", base)
+    visto_df = _contar(df, "select count(*) from public.vistoria")
